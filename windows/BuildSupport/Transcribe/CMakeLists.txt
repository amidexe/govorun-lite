cmake_minimum_required(VERSION 3.25)
project(govorun_transcribe VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS ON)
set(TRANSCRIBE_BUILD_SHARED ON)
set(TRANSCRIBE_GGML_BACKEND_DL ON)
set(TRANSCRIBE_USE_SYSTEM_BLAS ON)
set(TRANSCRIBE_BUILD_COMMIT "ea077b8-govorun-blas1")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_REQUIRED_INCLUDES "${OPENBLAS_ROOT}/include")
foreach(name ggml-base ggml)
  add_library(${name} SHARED IMPORTED GLOBAL)
  set_target_properties(${name} PROPERTIES
    IMPORTED_IMPLIB "${RUNTIME_ROOT}/${name}.lib"
    IMPORTED_LOCATION "${RUNTIME_ROOT}/${name}.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${TRANSCRIBE_SOURCE}/ggml/include"
    INTERFACE_COMPILE_DEFINITIONS "GGML_SHARED")
endforeach()
set_property(TARGET ggml PROPERTY INTERFACE_LINK_LIBRARIES ggml-base)
function(transcribe_apply_warnings target)
  target_compile_options(${target} PRIVATE /W3 /EHs)
endfunction()
add_subdirectory("${TRANSCRIBE_SOURCE}/src" transcribe)
if(NOT HAVE_CBLAS_H)
  message(FATAL_ERROR "OpenBLAS C interface is required; scalar decoder builds are not supported")
endif()
target_include_directories(transcribe PUBLIC "${TRANSCRIBE_SOURCE}/include" PRIVATE "${OPENBLAS_ROOT}/include")
